{{- /* Turns an entry page into the entry object the templates and scripts share. Whatever the front matter leaves out is "not assessed", so nothing is ever invented. */ -}} {{- $p := .Params -}} {{- $fw := hugo.Data.framework -}} {{- $elig := dict -}} {{- range $fw.eligibility -}}{{- $elig = merge $elig (dict .key (index (default dict $p.eligibility) .key | default "not_verified")) -}}{{- end -}} {{- $prin := dict -}} {{- range $fw.principles -}}{{- $prin = merge $prin (dict .id (index (default dict $p.principles) .id)) -}}{{- end -}} {{- $complete := eq $p.assessment_status "complete" -}} {{- return (dict "id" .File.ContentBaseName "name" .Title "category" $p.category "summary" $p.summary "substrate" $p.substrate "syncModel" $p.sync_model "extractCmd" $p.extract_cmd "limitations" (default slice $p.limitations) "assessment_batch" $p.assessment_batch "pilot_stage" $p.pilot_stage "version_assessed" $p.version_assessed "assessment_date" $p.assessment_date "assessment_status" (cond $complete "complete" "unverified") "assessment_notes" (default "This entry is a candidate for assessment. No assessment under the evaluation framework has been performed yet, so no findings or determination are published." $p.assessment_notes) "eligibility" $elig "principles" $prin "evidence" (default slice $p.evidence) "determination" (cond $complete "computed" "insufficient_evidence") ) -}}
